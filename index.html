/* ===================================== 1. BIRTHDAY DATE ===================================== */ // November 18, 2026 // JavaScript months start at 0, so November = 10. const birthday = new Date(2026, 10, 18, 0, 0, 0); /* ===================================== 2. PAGE NAVIGATION ===================================== */ const pages = [ "countdownPage", "birthdayPage", "menuPage", "letterPage", "cakePage" ]; function showPage(pageId) { pages.forEach(id => { document.getElementById(id).classList.add("hidden"); }); document.getElementById(pageId).classList.remove("hidden"); window.scrollTo({ top: 0, behavior: "smooth" }); } /* ===================================== 3. COUNTDOWN ===================================== */ let birthdayUnlocked = false; function updateCountdown() { const now = new Date(); const difference = birthday - now; if (difference <= 0) { if (!birthdayUnlocked) { birthdayUnlocked = true; showPage("birthdayPage"); } clearInterval(countdownInterval); return; } const days = Math.floor(difference / 86400000); const hours = Math.floor((difference / 3600000) % 24); const minutes = Math.floor((difference / 60000) % 60); const seconds = Math.floor((difference / 1000) % 60); document.getElementById("days").textContent = days; document.getElementById("hours").textContent = hours; document.getElementById("minutes").textContent = minutes; document.getElementById("seconds").textContent = seconds; } updateCountdown(); const countdownInterval = setInterval(updateCountdown, 1000); /* ===================================== 4. BIRTHDAY HOMEPAGE ===================================== */ document.getElementById("openButton").addEventListener("click", () => { showPage("menuPage"); }); /* ===================================== 5. SURPRISE MENU ===================================== */ document.getElementById("letterButton").addEventListener("click", () => { showPage("letterPage"); }); document.getElementById("cakeButton").addEventListener("click", () => { showPage("cakePage"); }); /* ===================================== 6. BACK BUTTONS ===================================== */ document.querySelectorAll(".backButton").forEach(button => { button.addEventListener("click", () => { showPage("menuPage"); }); }); /* ===================================== 7. CREATE 17 BIRTHDAY CANDLES ===================================== */ const candlesContainer = document.getElementById("candles"); for (let i = 0; i < 17; i++) { const candle = document.createElement("div"); candle.className = "candle"; const flame = document.createElement("div"); flame.className = "flame"; candle.appendChild(flame); candlesContainer.appendChild(candle); } /* ===================================== 8. BLOW OUT THE CANDLES ===================================== */ const blowButton = document.getElementById("blowButton"); const resetButton = document.getElementById("resetButton"); blowButton.addEventListener("click", () => { // Turn off all 17 flames document.querySelectorAll(".flame").forEach(flame => { flame.classList.add("off"); }); // Change the message document.getElementById("cakeMessage").textContent = "You made a wish! ♡"; // Show celebration document.getElementById("celebration").classList.remove("hidden"); // Prevent repeated clicks until reset blowButton.disabled = true; // Create confetti createConfetti(); }); /* ===================================== 9. RESET THE CAKE ===================================== */ resetButton.addEventListener("click", () => { // Turn all flames back on document.querySelectorAll(".flame").forEach(flame => { flame.classList.remove("off"); }); // Restore original message document.getElementById("cakeMessage").textContent = "17 little candles, one big wish."; // Hide celebration document.getElementById("celebration").classList.add("hidden"); // Enable blow button again blowButton.disabled = false; }); /* ===================================== 10. CONFETTI ANIMATION ===================================== */ function createConfetti() { const decorations = ["🎉", "✨", "💗", "🎊", "♡"]; for (let i = 0; i < 60; i++) { const confetti = document.createElement("span"); confetti.textContent = decorations[Math.floor(Math.random() * decorations.length)]; confetti.style.position = "fixed"; confetti.style.left = Math.random() * 100 + "vw"; confetti.style.top = "-30px"; confetti.style.fontSize = (16 + Math.random() * 15) + "px"; confetti.style.pointerEvents = "none"; confetti.style.zIndex = "9999"; confetti.style.animation = `fall ${2 + Math.random() * 3}s linear forwards`; document.body.appendChild(confetti); setTimeout(() => { confetti.remove(); }, 5000); } }
